function Orders() { const [orders, setOrders] = useState([]); const [message, setMessage] = useState(''); const load = () => api('/api/admin/orders').then(setOrders).catch((error) => setMessage(error.message)); useEffect(() => { load() }, []); const update = async (id, status) => { try { await api(`/api/admin/orders/${id}/status`, { method: 'PUT', body: JSON.stringify({ status }) }); load() } catch (error) { setMessage(error.message) } }; return <div className="content-wrap"><PageIntro eyebrow="Admin workspace · Fulfillment" title="Order and shipping history." text="Track every order from pending to delivered." count={orders.length} />{message && <div className="auth-message">{message}</div>}<section className="panel table-panel">{orders.length ? <table><thead><tr><th>Order</th><th>Customer</th><th>Total</th><th>Status</th><th>Update</th></tr></thead><tbody>{orders.map((order) => <tr key={order.id}><td><strong>#{order.id.slice(0, 8)}</strong><small>{new Date(order.createdAt).toLocaleString()}</small></td><td>{order.customerName}<small>{order.customerEmail}</small></td><td>PHP {Number(order.total).toLocaleString()}</td><td><span className="status-pill">{order.status}</span></td><td><select value={order.status} onChange={(event) => update(order.id, event.target.value)}><option>Pending</option><option>Processing</option><option>Shipped</option><option>Delivered</option><option>Cancelled</option></select></td></tr>)}</tbody></table> : <div className="empty-shelf"><Truck size={27} /><h3>No orders yet.</h3><p>Shipping history will appear here when buyer orders are connected.</p></div>}</section></div> }
