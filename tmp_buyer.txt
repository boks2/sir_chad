function BuyerWorkspace({ onLogout }) { const [view, setView] = useState('shop'); const [products, setProducts] = useState([]); const [cart, setCart] = useState([]); const [profile, setProfile] = useState({ name: '', email: '', phone: '', address: '' }); const [orders, setOrders] = useState([]); const [message, setMessage] = useState(''); const [payment, setPayment] = useState('Online Payment'); const [query, setQuery] = useState(''); const [category, setCategory] = useState('All categories'); const [sortBy, setSortBy] = useState('featured'); const persistCart = async (items) => { try { await api('/api/buyer/cart', { method: 'PUT', body: JSON.stringify({ items }) }) } catch (error) { console.warn('Cart sync failed:', error.message) } }; useEffect(() => { api('/api/products').then(setProducts).catch((error) => setMessage(error.message)); api('/api/buyer/profile').then(setProfile).catch(() => {}); api('/api/buyer/orders').then(setOrders).catch(() => setOrders([])); api('/api/buyer/cart').then((data) => setCart(Array.isArray(data.items) ? data.items : [])).catch(() => setCart([])) }, []); const categories = ['All categories', 'Pod Systems', 'E-liquids', 'Disposables', 'Accessories', 'Coils']; const filteredProducts = products.filter((product) => (!query || [product.name, product.category, product.flavor, product.description].join(' ').toLowerCase().includes(query.toLowerCase())) && (category === 'All categories' || product.category === category)).sort((a, b) => sortBy === 'price-low' ? Number(a.price) - Number(b.price) : sortBy === 'price-high' ? Number(b.price) - Number(a.price) : sortBy === 'name' ? a.name.localeCompare(b.name) : 0); const addToCart = (product) => { setCart((current) => { const existing = current.find((item) => item.id === product.id); const next = existing ? current.map((item) => item.id === product.id ? { ...item, quantity: item.quantity + 1 } : item) : [...current, { ...product, quantity: 1 } ]; persistCart(next); return next }); }; const changeQuantity = (id, quantity) => { setCart((current) => { const next = current.map((item) => item.id === id ? { ...item, quantity: Math.max(1, Math.min(Number(quantity), item.stock)) } : item); persistCart(next); return next }); }; const total = cart.reduce((sum, item) => sum + Number(item.price) * item.quantity, 0); const checkout = async () => { setMessage(''); const selectedPayment = String(payment || '').trim(); const serverCart = await api('/api/buyer/cart').then((data) => Array.isArray(data.items) ? data.items : []).catch(() => cart); const checkoutItems = serverCart.length ? serverCart : cart; if (!checkoutItems.length) { setMessage('Your cart is empty. Add a product before checkout.'); return } const buyerReady = Boolean(profile.name && profile.email && profile.phone && profile.address); if (!buyerReady) { setMessage('Complete your buyer information before checkout.'); setView('profile'); return } try { if (selectedPayment === 'Online Payment') { const order = await api('/api/buyer/orders', { method: 'POST', body: JSON.stringify({ items: checkoutItems, paymentMethod: 'Online Payment' }) }); const invoice = await api('/api/payments/xendit/invoice', { method: 'POST', body: JSON.stringify({ amount: Number(order.total), description: `Vaporly order ${order.id.slice(0, 8)}`, externalId: order.id }) }); setCart([]); await api('/api/buyer/cart', { method: 'PUT', body: JSON.stringify({ items: [] }) }); setView('orders'); setOrders((current) => [order, ...current]); setMessage('Xendit checkout opened in a new tab. Complete the test payment to finish your order.'); window.open(invoice.invoiceUrl, '_blank', 'noopener,noreferrer'); return } const data = await api('/api/buyer/orders', { method: 'POST', body: JSON.stringify({ items: checkoutItems, paymentMethod: 'COD' }) }); setCart([]); await api('/api/buyer/cart', { method: 'PUT', body: JSON.stringify({ items: [] }) }); setView('orders'); setOrders((current) => [data, ...current]); setMessage('Order placed successfully. The shop will process your delivery.') } catch (error) { setMessage(error.message) } }; const saveProfile = async (event) => { event.preventDefault(); try { const data = await api('/api/buyer/profile', { method: 'PUT', body: JSON.stringify(profile) }); setProfile(data); setMessage('Profile saved. You can now place your order.') } catch (error) { setMessage(error.message) } }; const filtered = products.filter((product) => !query || [product.name, product.category, product.flavor].join(' ').toLowerCase().includes(query.toLowerCase())); return <div className="buyer-shell"><header className="buyer-header"><Logo /><div className="buyer-actions"><button className={view === 'shop' ? 'active' : ''} onClick={() => setView('shop')}>Shop</button><button className={view === 'orders' ? 'active' : ''} onClick={() => { setView('orders'); api('/api/buyer/orders').then(setOrders).catch(() => {}) }}>My orders</button><button className={view === 'profile' ? 'active' : ''} onClick={() => setView('profile')}>My settings</button><button className="cart-button" onClick={() => setView('cart')}><ShoppingCart size={17} /> Cart ({cart.reduce((sum, item) => sum + item.quantity, 0)})</button><button className="text-button" onClick={onLogout}>Log out</button></div></header><main className="buyer-content">{message && <div className="auth-message">{message}</div>}{view === 'shop' && <><div className="buyer-hero"><p className="eyebrow">Vaporly buyer shop</p><h1>Find your next favorite.</h1><p>Browse products added by the store and add them to your cart before checkout.</p></div><div className="search-bar"><Search size={18} /><input value={query} onChange={(event) => setQuery(event.target.value)} placeholder="Search products..." /></div><div className="product-grid buyer-product-grid">{filtered.map((product) => <article className="product-card" key={product.id}><div className="product-art">{product.image ? <img src={product.image} alt={product.name} /> : <PackageSearch size={38} />}</div><div className="product-body"><span className="product-category">{product.category}</span><h3>{product.name}</h3><div className="product-meta"><strong>PHP {Number(product.price).toLocaleString()}</strong><span>{product.stock} left</span></div><button className="button primary full" onClick={() => addToCart(product)} disabled={!product.stock}>Add to cart</button></div></article>)}</div>{!filtered.length && <div className="empty-shelf"><PackageSearch size={27} /><h3>No products available yet.</h3><p>The store will display products here after admin adds them.</p></div>}</>}{view === 'cart' && <section className="buyer-panel"><PageIntro eyebrow="Buyer checkout" title="Your cart." text="Review quantities and choose your payment method before ordering." count={cart.length} />{cart.length ? <><div className="cart-list">{cart.map((item) => <div className="cart-row" key={item.id}><img src={item.image} alt={item.name} /><span><strong>{item.name}</strong><small>PHP {Number(item.price).toLocaleString()} each</small></span><input type="number" min="1" max={item.stock} value={item.quantity} onChange={(event) => changeQuantity(item.id, event.target.value)} /><strong>PHP {(Number(item.price) * item.quantity).toLocaleString()}</strong></div>)}</div><div className="checkout-box"><h3>Payment method</h3><select value={payment} onChange={(event) => setPayment(event.target.value)}><option value="Online Payment">Online Payment</option><option value="COD">COD</option></select><strong>Total: PHP {total.toLocaleString()}</strong><button className="button primary" onClick={checkout}>Checkout <ArrowRight size={16} /></button><small>Your buyer profile must have complete contact and delivery information.</small></div></> : <div className="empty-shelf"><ShoppingCart size={27} /><h3>Your cart is empty.</h3><p>Add products from the shop before checkout.</p></div>}</section>}{view === 'profile' && <form className="settings-card buyer-profile" onSubmit={saveProfile}><PageIntro eyebrow="Buyer settings" title="Your information." text="Complete this profile before placing an order." /><label>Full name<input value={profile.name} onChange={(event) => setProfile({ ...profile, name: event.target.value })} required /></label><label>Gmail address<input type="email" value={profile.email} onChange={(event) => setProfile({ ...profile, email: event.target.value })} required /></label><label>Phone number<input value={profile.phone} onChange={(event) => setProfile({ ...profile, phone: event.target.value })} required /></label><label>Delivery address<textarea value={profile.address} onChange={(event) => setProfile({ ...profile, address: event.target.value })} required rows="4" /></label><button className="button primary">Save buyer information <ArrowRight size={16} /></button></form>}{view === 'orders' && <section className="buyer-panel"><PageIntro eyebrow="Buyer account" title="My orders." text="Track the status of your orders." count={orders.length} />{orders.length ? orders.map((order) => <div className="order-card" key={order.id}><div><strong>Order #{order.id.slice(0, 8)}</strong><small>{new Date(order.createdAt).toLocaleString()}</small></div><span className="status-pill">{order.status}</span><strong>PHP {Number(order.total).toLocaleString()}</strong></div>) : <div className="empty-shelf"><History size={27} /><h3>No orders yet.</h3><p>Your placed orders will appear here.</p></div>}</section>}</main></div> }
